💻 使用方法

JavaScript での使用例

import { NinjaTable } from 'ninja-table';

async function initTable() {
  const canvas = document.getElementById('myCanvas');
  
  // テーブルを作成
  const table = await NinjaTable.create(canvas, {
    row_count: 50,
    col_count: 10,
    default_col_width: 120,
    default_row_height: 28
  });

  // 列ヘッダーを設定
  const columnHeaders = [
    {
      name: "id",
      display_name: "ID",
      width: 60,
      required: true,
      order: 0,
      is_visible: true,
      field_type: "IntegerField",
      min_number: 1,
      max_number: 99999
    },
    {
      name: "name",
      display_name: "名前",
      width: 150,
      required: true,
      order: 1,
      is_visible: true,
      field_type: "CharField",
      max_length: 50
    },
    {
      name: "email",
      display_name: "メールアドレス",
      width: 200,
      required: false,
      order: 2,
      is_visible: true,
      field_type: "EmailField",
      max_length: 100
    },
    {
      name: "salary",
      display_name: "給与",
      width: 100,
      required: false,
      order: 3,
      is_visible: true,
      field_type: "DecimalField",
      max_digits: 8,
      decimal_places: 0
    }
  ];

  // ヘッダー設定を適用
  table.setColumnHeaders(JSON.stringify(columnHeaders));

  // セルに値を設定
  table.setCellValue(0, 0, '1001');
  table.setCellValue(0, 1, '田中太郎');
  table.setCellValue(0, 2, 'tanaka@example.com');
  table.setCellValue(0, 3, '500000');

  // イベントハンドラーを設定
  table.setEventHandlers({
    onCellSelect: (position) => {
      console.log(`選択: ${NinjaTable.getCellReference(position.row, position.col)}`);
    }
  });

  // レンダリング
  table.render();
}

initTable();

TypeScript での使用例

import { 
  NinjaTable, 
  TableConfig, 
  CellPosition, 
  EventHandlers,
  ColumnHeader,
  FieldType 
} from 'ninja-table';

class SpreadsheetApp {
  private table: NinjaTable | null = null;

  async initialize(canvas: HTMLCanvasElement): Promise<void> {
    const config: Partial<TableConfig> = {
      row_count: 100,
      col_count: 10,
      default_col_width: 120,
      default_row_height: 28,
      font_family: 'Arial, sans-serif',
      font_size: 12
    };

    this.table = await NinjaTable.create(canvas, config);

    // 型安全な列ヘッダー設定
    const columnHeaders: ColumnHeader[] = [
      {
        name: "employee_id",
        display_name: "社員ID",
        width: 80,
        required: true,
        order: 0,
        is_visible: true,
        field_type: FieldType.IntegerField,
        min_number: 1,
        max_number: 99999
      },
      {
        name: "name",
        display_name: "氏名",
        width: 150,
        required: true,
        order: 1,
        is_visible: true,
        field_type: FieldType.CharField,
        max_length: 50
      },
      {
        name: "department",
        display_name: "部署",
        width: 120,
        required: true,
        order: 2,
        is_visible: true,
        field_type: FieldType.MenuField,
        choices: ["開発部", "営業部", "総務部", "人事部"]
      },
      {
        name: "salary",
        display_name: "月給",
        width: 100,
        required: false,
        order: 3,
        is_visible: true,
        field_type: FieldType.DecimalField,
        max_digits: 8,
        decimal_places: 0
      },
      {
        name: "active",
        display_name: "在籍中",
        width: 80,
        required: true,
        order: 4,
        is_visible: true,
        field_type: FieldType.BooleanField
      }
    ];

    // ヘッダー設定を適用
    this.table.setColumnHeaders(JSON.stringify(columnHeaders));

    const handlers: EventHandlers = {
      onCellSelect: this.handleCellSelect.bind(this),
      onCellChange: this.handleCellChange.bind(this)
    };

    this.table.setEventHandlers(handlers);
    this.table.render();
  }

  private handleCellSelect(position: CellPosition): void {
    console.log(`セル選択: ${position.row}, ${position.col}`);
  }

  private handleCellChange(position: CellPosition, value: string): void {
    console.log(`セル変更: ${position.row}, ${position.col} = ${value}`);
  }

  // フィールドタイプに応じた入力検証
  validateCellValue(colIndex: number, value: string): boolean {
    const headers = this.getColumnHeaders();
    const header = headers[colIndex];
    
    if (!header) return true;

    switch (header.field_type) {
      case FieldType.IntegerField:
        const num = parseInt(value);
        return !isNaN(num) && 
               (!header.min_number || num >= header.min_number) &&
               (!header.max_number || num <= header.max_number);
               
      case FieldType.EmailField:
        return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value);
        
      case FieldType.CharField:
        return !header.max_length || value.length <= header.max_length;
        
      case FieldType.BooleanField:
        return ['true', 'false', '1', '0', 'yes', 'no'].includes(value.toLowerCase());
        
      default:
        return true;
    }
  }

  private getColumnHeaders(): ColumnHeader[] {
    if (!this.table) return [];
    const headersJson = this.table.getColumnHeaders();
    return JSON.parse(headersJson);
  }
}

React での使用例

import React, { useEffect, useRef, useState } from 'react';
import { NinjaTable } from 'ninja-table';

const NinjaTableComponent = () => {
  const canvasRef = useRef(null);
  const [table, setTable] = useState(null);
  const [selectedCell, setSelectedCell] = useState('A1');

  useEffect(() => {
    const initTable = async () => {
      if (canvasRef.current) {
        const tableInstance = await NinjaTable.create(canvasRef.current, {
          row_count: 50,
          col_count: 10,
          default_col_width: 120,
          default_row_height: 28
        });

        tableInstance.setEventHandlers({
          onCellSelect: (position) => {
            const cellRef = NinjaTable.getCellReference(position.row, position.col);
            setSelectedCell(cellRef);
          }
        });

        tableInstance.render();
        setTable(tableInstance);
      }
    };

    initTable();
  }, []);

  return (
    <div>
      <div>選択セル: {selectedCell}</div>
      <canvas ref={canvasRef} width={800} height={400} />
    </div>
  );
};

export default NinjaTableComponent;

列ヘッダー設定

NinjaTableでは、各列に詳細な設定を行うことができます。

📋 ColumnHeader設定項目

{
  name: string,              // プロパティ名（内部識別用）
  display_name: string,      // 表示名（ヘッダーに表示される名前）
  width: number,             // 列の幅（ピクセル）
  required: boolean,         // 必須入力項目かどうか
  order: number,             // 表示順序
  is_visible: boolean,       // 表示/非表示フラグ
  field_type: FieldType,     // フィールドタイプ
  max_length?: number,       // 最大文字数（文字列フィールド用）
  max_digits?: number,       // 整数桁数（数値フィールド用）
  decimal_places?: number,   // 小数点桁数（小数フィールド用）
  min_number?: number,       // 最小値（数値フィールド用）
  max_number?: number,       // 最大値（数値フィールド用）
  choices?: string[]         // 選択肢（メニューフィールド用）
}

🎯 フィールドタイプ一覧

  • CharField - 短いテキスト入力（名前、コードなど）
  • EmailField - メールアドレス入力（検証付き）
  • TextareaField - 長いテキスト入力（備考、説明など）
  • IntegerField - 整数値入力（年齢、個数など）
  • DecimalField - 小数点数値入力（金額、税率など）
  • DecimalWithNullField - NULL許可の小数点数値
  • DateField - 日付入力（生年月日、入社日など）
  • TimeField - 時刻入力（勤務時間など）
  • BooleanField - チェックボックス（有効/無効など）
  • ButtonField - ボタン表示（アクション実行用）
  • MenuField - ドロップダウンメニュー（選択肢から選択）

✨ 主な機能

🚀 高性能レンダリング

Rust + WebAssemblyによる最適化されたパフォーマンス。大規模データもスムーズに処理。

📊 Excel風操作

キーボードナビゲーション、セル編集、数式バーなど、Excelライクな操作感。

🎯 TypeScript対応

完全な型安全性とIntelliSenseサポート。JavaScript/TypeScript両方で利用可能。

🎨 カスタマイズ可能

豊富な設定オプションで、デザインや動作を自由にカスタマイズ。

📱 レスポンシブ

モダンブラウザ対応。WebAssemblyをサポートする環境で動作。

🪶 軽量

最小限の依存関係で、バンドルサイズを抑制。

🎮 操作方法

⌨️ キーボード操作

  • 矢印キー: セル移動
  • Enter: 編集開始 / 下のセルに移動
  • F2: 編集開始（既存値保持）
  • Tab: 右のセルに移動
  • Escape: 編集キャンセル
  • Delete/Backspace: セル内容削除

🖱️ マウス操作

  • クリック: セル選択
  • ホイール: スクロール
  • ダブルクリック: 編集開始（予定）
